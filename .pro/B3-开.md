- 改 `query.rs:31-37`（`Some(t) if t < threshold` → `t <= threshold`），注释改成"expires_at 是失效时刻（含端点）"。

### 3.3 `store.rs` 新增三个方法（放在 `set_expiry` 后面，现 366-373 之后）

```rust
/// 把 subject 下除 keep 以外的判断全部标记过期（expires_at = at，含端点即失效）。
/// 返回被改的行数；已过期的不重复写。
pub fn expire_older_than(&self, subject: &str, keep: Option<&str>, at: i64) -> Result<usize> {
    let n = self.conn.execute(
        "UPDATE judgment SET expires_at=?1
         WHERE subject=?2 AND (?3 IS NULL OR id<>?3) AND (expires_at IS NULL OR expires_at>?1)",
        params![at, subject, keep],
    )?;
    Ok(n)
}

/// 该 subject 上一条未解决的冲突（judgment_a, judgment_b）。
pub fn unresolved_conflict_for(&self, subject: &str) -> Result<Option<(String, String)>> {
    let row = self.conn.query_row(
        "SELECT c.judgment_a, c.judgment_b FROM conflict c
         JOIN judgment a ON a.id=c.judgment_a
         JOIN judgment b ON b.id=c.judgment_b
         WHERE c.resolved=0 AND a.subject=?1 AND b.subject=?1 LIMIT 1",
        params![subject], |r| Ok((r.get(0)?, r.get(1)?)),
    ).optional()?;
    Ok(row)
}

/// 标记一对冲突已解决（id 与 put_conflict 同格式：cf:a:b）。
pub fn mark_conflict_resolved(&self, a: &str, b: &str, at: i64) -> Result<()> {
    self.conn.execute(
        "UPDATE conflict SET resolved=1, resolved_at=?1 WHERE id=?2",
        params![at, format!("cf:{a}:{b}")],
    )?;
    Ok(())
}
```

```rust
fn cmd_expire(root: &PathBuf, subject: &str, at: Option<i64>, id: Option<String>) -> anyhow::Result<ExitCode> {
    let store = Store::open(root)?;
    let ts = at.unwrap_or_else(datacore::store::now_secs);
    let target = match id {
        Some(id) => {
            let s = store.judgment_subject(&id)?.unwrap_or_default();
            if s != subject { anyhow::bail!("判断 {id} 不属于 {subject}"); }
            id
        }
        None => match store.find_judgment(subject)? {
            Some(j) => j.id,
            None => { println!("no_data: {subject}"); return Ok(ExitCode::from(2)); }
        },
    };
    store.set_expiry(&target, ts)?;
    println!("expired: {subject}（id …{}）at {ts}", &target[target.len().saturating_sub(8)..]);
    Ok(ExitCode::SUCCESS)
}
```

```rust
fn cmd_resolve(root: &PathBuf, subject: &str, verdict: &str, by: &str, why: &str) -> anyhow::Result<ExitCode> {
    let store = Store::open(root)?;
    let repo = store.repo().to_string();
    let now = datacore::store::now_secs();
    if by.trim().is_empty() || why.trim().is_empty() {
        anyhow::bail!("--by 与 --why 都不能为空（resolve 必须有人和理由）");
    }
    let v = match verdict {
        "held" => Verdict::Held,
        "refuted" => Verdict::Refuted,
        other => anyhow::bail!("--verdict 只接受 held|refuted，收到 {other}"),
    };
    let Some((a, b)) = store.unresolved_conflict_for(subject)? else {
        println!("no_conflict: {subject}");
        return Ok(ExitCode::from(2));
    };
    let Some(old) = store.find_judgment(subject)? else {
        println!("no_data: {subject}"); return Ok(ExitCode::from(2));
    };
    let j = judgment_row(&repo, subject, &old.kind, v, now, 1, 1, old.scope.clone());
    store.put_judgment(&j)?;

    let src = Source { /* instrument="resolve", env=format!("repo={repo};by={by}"), time=now */ };
    store.put_source(&src)?;
    let why120: String = why.chars().take(120).collect();   // clip 已随 issue/pr 删除，手写截断
    let premise = format!("resolve:cf:{}:{}:{}", a, b, why120);
    store.put_evidence(&Evidence {
        id: evidence_id(&j.id, &premise, if v == Verdict::Held { "supports" } else { "contradicts" }),
        judgment_id: j.id.clone(), premise,
        polarity: if v == Verdict::Held { Polarity::Supports } else { Polarity::Contradicts },
        source_id: src.id,
    })?;
    let n = store.expire_older_than(subject, Some(&j.id), now)?;
    store.mark_conflict_resolved(&a, &b, now)?;
    println!("resolved: {subject} -> {verdict} by {by}（旧判断过期 {n} 条）");
    Ok(ExitCode::SUCCESS)
}
```

> `clip` 在 B2.3 已随 issue/pr 删除；这里用 `why.chars().take(120).collect::<String>()`，别把整段理由塞进 premise（指针原则）。
